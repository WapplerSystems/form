# cat=Backend; type=boolean; label=Site sender feature: lets site administrators maintain email sender addresses in site configuration, and the form plugin's FlexForm provides a dropdown to pick one. When off the standard form-YAML senderAddress option is used.
featureSiteEmail = 0

# cat=Backend; type=boolean; label=Mail log: record one row per notification mail an Email finisher tries to send, so a failed or stuck send is visible instead of silent. Off keeps the extension byte-for-byte as before.
featureMailLog = 0

# cat=Backend; type=boolean; label=Mail log for every form: also record rows for forms that did not opt in. Those rows carry NO personal data (no recipient, subject or sender) - only form, finisher, status, error code and timestamps. Recommended on, because the form nobody watches is the form nobody opts in. Turn off for strict per-form opt-in.
mailLogAllForms = 1

# cat=Backend; type=boolean; label=Consent log: record one row per consent checkbox per submission (which consent, given or not, and a hash of the exact wording shown) so the consent can be demonstrated under Art. 7(1) GDPR independently of the notification mail. Needs a retention decision - see CleanupConsentLogTask.
featureConsentLog = 0

# cat=Backend; type=boolean; label=Automatic form shutdown: when a form is flooded, disable it by itself for a while instead of waiting for someone to notice. Off by default - it is an automation that takes a site feature away, so it should be a deliberate choice. Needs the validation log, which this flag switches on for every form (that log stores no submitted values). Threshold, window, duration and the notification address are set in the form log module.
featureFormAutoDisable = 0
